function showRandomCard() { const n = Math.floor(Math.random() * 78) + 1; const img = new Image(); img.src = `cards/${n}.png`; img.onload = () => { console.log(`Loaded card ${n}`); document.body.appendChild(img); }; img.onerror = () => { console.log(`Error loading card ${n}`); }; } showRandomCard();
